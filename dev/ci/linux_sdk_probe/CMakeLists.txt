cmake_minimum_required(VERSION 3.21)
project(LinuxInputProbe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_PREFIX_PATH "${QTROOT}")
find_package(Qt6 6.9.3 EXACT REQUIRED COMPONENTS Core Gui Network WebChannel Qml Quick Xml WebEngineCore WebEngineQuick Widgets OpenGL DBus Test Positioning)
find_package(OpenGL REQUIRED COMPONENTS OpenGL EGL)
find_library(GLES2_LIBRARY NAMES GLESv2 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(XKB REQUIRED IMPORTED_TARGET xkbcommon)
foreach(module IN ITEMS Core Gui Network WebChannel Qml Quick Xml WebEngineCore WebEngineQuick Widgets OpenGL DBus Test Positioning)
  get_target_property(location Qt6::${module} IMPORTED_LOCATION_RELEASE)
  if(NOT location)
    get_target_property(location Qt6::${module} IMPORTED_LOCATION)
  endif()
  file(REAL_PATH "${location}" actual)
  file(REAL_PATH "${QTROOT}" selected)
  cmake_path(IS_PREFIX selected "${actual}" NORMALIZE within)
  if(NOT within)
    message(FATAL_ERROR "Qt6::${module} resolves outside selected prefix: ${actual}")
  endif()
  list(APPEND qt_targets Qt6::${module})
endforeach()
if(NOT TARGET Qt6::GuiPrivate OR NOT TARGET Qt6::QuickPrivate OR NOT TARGET Qt6::WebEngineCorePrivate)
  message(FATAL_ERROR "Qt private development targets missing")
endif()
add_executable(qt_mpv_probe probe.cpp ../mpv_link_probe.c)
set_source_files_properties(../mpv_link_probe.c PROPERTIES COMPILE_DEFINITIONS "main=mpv_probe_main")
target_include_directories(qt_mpv_probe PRIVATE "${MPV_INCLUDE_DIR}")
target_link_libraries(qt_mpv_probe PRIVATE ${qt_targets} Qt6::GuiPrivate Qt6::QuickPrivate Qt6::WebEngineCorePrivate OpenGL::OpenGL OpenGL::EGL "${GLES2_LIBRARY}" PkgConfig::XKB "${MPV_LIBRARY}" ${CMAKE_DL_LIBS})
