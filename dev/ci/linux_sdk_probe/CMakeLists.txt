cmake_minimum_required(VERSION 3.21)
project(LinuxInputProbe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_PREFIX_PATH "${QTROOT}")
include("${CMAKE_CURRENT_LIST_DIR}/probe_checks.cmake")
find_package(Qt6 6.9.3 EXACT REQUIRED COMPONENTS Core Gui Network WebChannel Qml Quick Xml WebEngineCore WebEngineQuick Widgets OpenGL DBus Test Positioning)
find_package(OpenGL QUIET)
if(NOT OPENGL_FOUND)
  list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/../../../CMakeModules")
  find_package(GLES2 QUIET)
endif()
select_graphics("${OPENGL_FOUND}" "${OPENGL_gl_LIBRARY}" "${GLES2_FOUND}" "${GLES2_LIBRARY}")
find_package(PkgConfig REQUIRED)
pkg_check_modules(XKB REQUIRED IMPORTED_TARGET xkbcommon)
foreach(module IN ITEMS Core Gui Network WebChannel Qml Quick Xml WebEngineCore WebEngineQuick Widgets OpenGL DBus Test Positioning)
  require_qt_target_location(Qt6::${module})
  list(APPEND qt_targets Qt6::${module})
endforeach()
foreach(target IN ITEMS Qt6::GuiPrivate Qt6::QuickPrivate Qt6::WebEngineCorePrivate)
  if(NOT TARGET ${target})
    message(FATAL_ERROR "Missing Qt private target: ${target}")
  endif()
  string(REPLACE "::" "_" name "${target}")
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/${name}-includes.txt"
    CONTENT "$<JOIN:$<TARGET_GENEX_EVAL:${target},$<TARGET_PROPERTY:${target},INTERFACE_INCLUDE_DIRECTORIES>>,\n>")
endforeach()
foreach(target IN ITEMS Qt6::moc Qt6::rcc Qt6::uic Qt6::qmlimportscanner Qt6::qmlcachegen Qt6::qmltyperegistrar)
  require_qt_target_location(${target})
endforeach()
add_executable(qt_mpv_probe probe.cpp ../mpv_link_probe.c)
set_source_files_properties(../mpv_link_probe.c PROPERTIES COMPILE_DEFINITIONS "main=mpv_probe_main")
target_include_directories(qt_mpv_probe PRIVATE "${MPV_INCLUDE_DIR}")
if(PROBE_GRAPHICS_DEFINE STREQUAL "PROBE_GLES2")
  target_include_directories(qt_mpv_probe PRIVATE "${GLES2_INCLUDE_DIR}")
else()
  target_include_directories(qt_mpv_probe PRIVATE "${OPENGL_INCLUDE_DIR}")
endif()
target_compile_definitions(qt_mpv_probe PRIVATE ${PROBE_GRAPHICS_DEFINE})
target_link_libraries(qt_mpv_probe PRIVATE ${qt_targets} Qt6::GuiPrivate Qt6::QuickPrivate Qt6::WebEngineCorePrivate "${PROBE_GRAPHICS_LIB}" PkgConfig::XKB "${MPV_LIBRARY}" ${CMAKE_DL_LIBS})
