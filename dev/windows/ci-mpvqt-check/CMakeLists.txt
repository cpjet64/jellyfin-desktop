cmake_minimum_required(VERSION 3.21)
project(CheckJobMpvQt LANGUAGES CXX)
find_package(MpvQt CONFIG REQUIRED)
get_target_property(_configs MpvQt::MpvQt IMPORTED_CONFIGURATIONS)
set(_found FALSE)
foreach(_config IN LISTS _configs)
  get_target_property(_location MpvQt::MpvQt "IMPORTED_LOCATION_${_config}")
  if(_location)
    file(REAL_PATH "${_location}" _real)
    file(REAL_PATH "${EXPECTED_MPVQT_PREFIX}" _prefix)
    cmake_path(IS_PREFIX _prefix "${_real}" NORMALIZE _inside)
    if(NOT _inside OR NOT EXISTS "${_real}")
      message(FATAL_ERROR "MpvQt imported runtime outside job-local prefix: ${_real}")
    endif()
    message(STATUS "Verified imported MpvQt runtime: ${_real}")
    set(_found TRUE)
  endif()
endforeach()
if(NOT _found)
  message(FATAL_ERROR "No installed MpvQt runtime found")
endif()
