find_package(Qt6 6.9.3 EXACT REQUIRED COMPONENTS Core Gui Quick OpenGL)
if(NOT TARGET Qt6::QuickPrivate OR NOT TARGET Qt6::GuiPrivate)
  message(FATAL_ERROR "Qt 6.9.3 matching QuickPrivate and GuiPrivate targets are required")
endif()
add_library(qt_threaded_fixture STATIC helper.cpp cleanup.cpp)
get_filename_component(QT_THREADED_PREFIX "${Qt6_DIR}/../../.." REALPATH)
foreach(component IN ITEMS Core Gui Quick)
  get_filename_component(component_prefix "${Qt6${component}_DIR}/../../.." REALPATH)
  if(NOT component_prefix STREQUAL QT_THREADED_PREFIX)
    message(FATAL_ERROR "Qt ${component} package does not share the Qt private-header prefix")
  endif()
  get_target_property(imported_configs Qt6::${component} IMPORTED_CONFIGURATIONS)
  set(found_imported_location FALSE)
  foreach(config IN LISTS imported_configs)
    get_target_property(location Qt6::${component} IMPORTED_LOCATION_${config})
    if(location AND NOT location MATCHES "-NOTFOUND$")
      set(found_imported_location TRUE)
      get_filename_component(resolved_location "${location}" REALPATH)
      cmake_path(IS_PREFIX QT_THREADED_PREFIX "${resolved_location}" NORMALIZE location_in_prefix)
      if(NOT location_in_prefix)
        message(FATAL_ERROR "Qt ${component} imported library lies outside the expected prefix")
      endif()
    endif()
  endforeach()
  if(NOT found_imported_location)
    message(FATAL_ERROR "Qt ${component} imported library location is unavailable")
  endif()
endforeach()
foreach(private_target IN ITEMS QuickPrivate GuiPrivate)
  get_target_property(private_includes Qt6::${private_target} INTERFACE_INCLUDE_DIRECTORIES)
  foreach(entry IN LISTS private_includes)
    if(entry MATCHES "(/include/|\\.framework/)")
      string(FIND "${entry}" "${QT_THREADED_PREFIX}/" prefix_position)
      if(prefix_position EQUAL -1)
        message(FATAL_ERROR "Qt ${private_target} private include lies outside the expected prefix")
      endif()
    endif()
  endforeach()
endforeach()
foreach(header IN ITEMS qquickrendercontrol_p.h qquickwindow_p.h)
  if(NOT EXISTS "${QT_THREADED_PREFIX}/include/QtQuick/6.9.3/QtQuick/private/${header}" AND
     NOT EXISTS "${QT_THREADED_PREFIX}/lib/QtQuick.framework/Headers/6.9.3/QtQuick/private/${header}")
    message(FATAL_ERROR "Missing matching Qt 6.9.3 private header: ${header}")
  endif()
endforeach()
if(NOT EXISTS "${QT_THREADED_PREFIX}/include/QtGui/6.9.3/QtGui/rhi/qrhi.h" AND
   NOT EXISTS "${QT_THREADED_PREFIX}/lib/QtGui.framework/Headers/6.9.3/QtGui/rhi/qrhi.h")
  message(FATAL_ERROR "Missing matching Qt 6.9.3 QRhi header")
endif()
target_compile_definitions(qt_threaded_fixture PRIVATE QT_THREADED_QT_PREFIX="${QT_THREADED_PREFIX}")
target_include_directories(qt_threaded_fixture PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(qt_threaded_fixture PUBLIC Qt6::Core Qt6::Gui Qt6::Quick Qt6::OpenGL
                                      PRIVATE Qt6::QuickPrivate Qt6::GuiPrivate)
