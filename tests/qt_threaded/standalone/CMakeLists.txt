cmake_minimum_required(VERSION 3.21)
project(QtThreadedFixture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.9.3 EXACT REQUIRED COMPONENTS Core Gui Quick OpenGL Test)
add_subdirectory(.. qt_threaded_helper)
add_executable(test_threaded_render ../test_threaded_render.cpp ../color_item.cpp)
target_link_libraries(test_threaded_render PRIVATE qt_threaded_fixture Qt6::Test)
enable_testing()

set_tests_properties(test_threaded_render PROPERTIES TIMEOUT 90)
